import React, { useState, useEffect } from 'react';
import { StyleSheet, Text, View, TouchableOpacity, Alert, SafeAreaView } from 'react-native';
import * as WebBrowser from 'expo-web-browser';
import * as AuthSession from 'expo-auth-session';
import { supabase } from './supabase';

// Essential for closing the login popup and returning to PopMatch
WebBrowser.maybeCompleteAuthSession();

export default function App() {
  const [user, setUser] = useState(null);

  useEffect(() => {
    // Check if a student is already signed in
    supabase.auth.getSession().then(({ data: { session } }) => {
      setUser(session?.user ?? null);
    });

    // Listen for sign-in/sign-out events
    const { data: authListener } = supabase.auth.onAuthStateChange((_event, session) => {
      setUser(session?.user ?? null);
    });

    return () => {
      authListener.subscription.unsubscribe();
    };
  }, []);

  const handleLogin = async () => {
    // Generates the return link for Expo Go on your phone
    const redirectTo = AuthSession.makeRedirectUri();
    
    // 1. Request the UIUC Microsoft Login URL from Supabase
    const { data, error } = await supabase.auth.signInWithOAuth({
      provider: 'azure',
      options: { 
        redirectTo, 
        scopes: 'email profile' 
      },
    });

    if (error) return Alert.alert("Login Error", error.message);

    // 2. Open the UIUC NetID login screen
    const result = await WebBrowser.openAuthSessionAsync(data.url, redirectTo);

    // 3. Complete the session when they return from the browser
    if (result.type === 'success') {
      const { params } = AuthSession.parseDeepLink(result.url);
      await supabase.auth.setSession({
        access_token: params.access_token,
        refresh_token: params.refresh_token,
      });
    }
  };

  return (
    <SafeAreaView style={styles.container}>
      <View style={styles.content}>
        {!user ? (
          <>
            <View style={styles.header}>
              <Text style={styles.logo}>PopMatch</Text>
              <Text style={styles.subtitle}>University of Illinois Urbana-Champaign</Text>
            </View>
            
            <TouchableOpacity style={styles.loginButton} onPress={handleLogin}>
              <Text style={styles.buttonText}>Sign in with NetID</Text>
            </TouchableOpacity>
            
            <Text style={styles.lockNote}>Verified @illinois.edu only</Text>
          </>
        ) : (
          <View style={styles.welcomeContainer}>
            <Text style={styles.welcomeHeader}>Welcome to PopMatch!</Text>
            <Text style={styles.userEmail}>{user.email}</Text>
            <TouchableOpacity 
              style={styles.logoutButton} 
              onPress={() => supabase.auth.signOut()}
            >
              <Text style={styles.logoutText}>Sign Out</Text>
            </TouchableOpacity>
          </View>
        )}
      </View>
    </SafeAreaView>
  );
}
